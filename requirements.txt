# clean 5.x install.

#              opencv-contrib-python, which is pinned below 5 for the same
#              reason as above -- an unpinned install pulls 5.x and takes the
#              cascade API away again.
#   pyaudio    microphone capture for acoustic event detection.
mediapipe>=0.10,<2
opencv-contrib-python>=4.8,<5
